Wanda and Kit

Interactive for Lecture 13: setting prices with few rivals

Wanda rents windsurf boards and Kit rents kiteboards on the same beach. Each rental costs its owner $10, and each charges \(H = \$36\) or \(L = \$20\). Each firm has 30 customers who prefer its sport: loyal ones rent from it at any price, price-sensitive ones pay \(H\) but switch if the other firm is cheaper, and 10 price-driven ones rent only at \(L\).

model = {
  const price = {H: 36, L: 20};
  const cost = 10;
  const label = {loyal: "loyal", sens: "price-sensitive", drive: "price-driven"};
  const other = f => f === "Wanda" ? "Kit" : "Wanda";
  // Where a group rents: its own firm, the other firm, or nobody (null).
  function where(owner, grp, p) {
    const mine = p[owner], theirs = p[other(owner)];
    if (grp === "loyal") return owner;
    if (grp === "sens") return (mine === "H" && theirs === "L") ? other(owner) : owner;
    return mine === "L" ? owner : (theirs === "L" ? other(owner) : null);
  }
  function blocks(l, p) {
    const n = {loyal: l, sens: 20 - l, drive: 10};
    const out = [];
    for (const owner of ["Wanda", "Kit"])
      for (const grp of ["loyal", "sens", "drive"])
        out.push({id: owner + "-" + grp, owner, grp, n: n[grp], at: where(owner, grp, p)});
    return out;
  }
  const count = (bs, firm) => bs.filter(b => b.at === firm).reduce((s, b) => s + b.n, 0);
  function pay(l, w, k) {
    const bs = blocks(l, {Wanda: w, Kit: k});
    return [(price[w] - cost) * count(bs, "Wanda"), (price[k] - cost) * count(bs, "Kit")];
  }
  return {price, cost, label, other, blocks, count, pay};
}
{
  const p = {Wanda: wanda, Kit: kit};
  const bs = model.blocks(loyal, p);
  const X = {Wanda: 20, Kit: 420}, unit = 18, bh = 36, gap = 8, y0 = 78;
  for (const firm of ["Wanda", "Kit"]) {
    // Own customers first, then any who walked over from the other stand.
    const here = bs.filter(b => b.at === firm && b.n > 0)
      .sort((a, b) => (a.owner === firm ? 0 : 1) - (b.owner === firm ? 0 : 1));
    here.forEach((b, i) => { b.x = X[firm]; b.y = y0 + i * (bh + gap); });
  }
  bs.filter(b => b.at === null && b.n > 0).forEach((b, i) => { b.x = 150 + i * 200; b.y = 391; });
  bs.filter(b => b.n === 0).forEach(b => { b.x = X[b.owner]; b.y = y0; });

  const solid = o => o === "Wanda" ? "#0072B2" : "#8f4100";
  const tint = o => o === "Wanda" ? "#dcebf6" : "#f6e4d6";
  const ink = o => o === "Wanda" ? "#005f94" : "#8f4100";
  const fill = b => b.grp === "loyal" ? solid(b.owner) : b.grp === "sens" ? "white" : tint(b.owner);
  const text = b => `${b.n} ${model.label[b.grp]}`;
  const inside = b => b.n * unit >= text(b).length * 8.6 + 12;
  const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  const svg = d3.select(stage);
  const t = svg.transition().duration(reduce ? 0 : 750);

  svg.select("#wk-blocks").selectAll("rect").data(bs, b => b.id)
    .join(enter => enter.append("rect").attr("x", b => b.x).attr("y", b => b.y)
                        .attr("width", b => b.n * unit).attr("height", bh).attr("rx", 3))
    .attr("fill", fill).attr("stroke", b => solid(b.owner)).attr("stroke-width", 2.5)
    .attr("stroke-dasharray", b => b.at === null ? "7 4" : null)
    .transition(t)
    .attr("x", b => b.x).attr("y", b => b.y).attr("width", b => b.n * unit)
    .attr("opacity", b => b.n > 0 ? 1 : 0);

  svg.select("#wk-labels").selectAll("text").data(bs, b => b.id)
    .join(enter => enter.append("text").attr("x", b => b.x).attr("y", b => b.y + bh / 2))
    .text(text)
    .attr("font-size", 17).attr("font-weight", 700).attr("dominant-baseline", "central")
    .attr("text-anchor", b => inside(b) ? "middle" : "start")
    .attr("fill", b => inside(b) && b.grp === "loyal" ? "white" : ink(b.owner))
    .transition(t)
    .attr("x", b => inside(b) ? b.x + b.n * unit / 2 : b.x + b.n * unit + 8)
    .attr("y", b => b.y + bh / 2)
    .attr("opacity", b => b.n > 0 ? 1 : 0);

  const parts = [];
  for (const firm of ["Wanda", "Kit"]) {
    const total = model.count(bs, firm);
    const incoming = bs.filter(b => b.at === firm && b.owner !== firm).reduce((s, b) => s + b.n, 0);
    const pr = model.price[p[firm]];
    const profit = (pr - model.cost) * total;
    const from = incoming ? ` (${incoming} from ${model.other(firm)})` : "";
    svg.select(`#wk-price-${firm}`).text(`charges ${p[firm]} = $${pr}`);
    svg.select(`#wk-total-${firm}`).text(`${total} customers${from}`);
    svg.select(`#wk-profit-${firm}`).text(`Profit: (${pr} − 10) × ${total} = ${profit}`);
    parts.push(`${firm} charges ${p[firm]}, $${pr}, and has ${total} customers${from}, for a profit of ${profit}.`);
  }
  const out = bs.filter(b => b.at === null).reduce((s, b) => s + b.n, 0);
  svg.select("#wk-out-label").attr("opacity", out ? 1 : 0);
  if (out) parts.push(`${out} price-driven customers do not rent.`);
  const summary = `With ${loyal} loyal customers per firm: ` + parts.join(" ");
  svg.attr("aria-label", "Customers at the two stands. " + summary);
  live.textContent = summary;
}
