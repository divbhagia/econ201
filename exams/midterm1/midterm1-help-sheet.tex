\newcommand{\hs}[1]{\par\vspace{14pt}\noindent{\large\bfseries\color{accent}#1}\par\vspace{-1pt}\noindent{\color{accent}\rule{\linewidth}{0.6pt}}\par\vspace{-1pt}}
% One table shape for every section: name, formula. Every row is built by
% \R (one line) or \F (a stacked fraction), which put the same strut in both
% cells, so the name sits on the formula's baseline, level with a fraction
% bar, and every row of a kind has the same height. (m columns would centre
% the cells, but under tagging=on they come out top-aligned.)
\newlength{\namecol}\setlength{\namecol}{3.5cm}
\newenvironment{sheettable}
  {\par\noindent\arrayrulecolor{rule}%
   \begin{tabular}{@{}>{\raggedright\arraybackslash}p{\namecol}@{\hspace{8pt}}>{\raggedright\arraybackslash}p{\dimexpr\linewidth-\namecol-8pt\relax}@{}}}
  {\end{tabular}\arrayrulecolor{black}\par}
% The height strut opens a cell and the depth strut closes it, so a cell
% that wraps keeps its normal line spacing.
\newcommand{\up}[1]{\rule{0pt}{#1}}
\newcommand{\dn}[1]{\rule[-#1]{0pt}{#1}}
\newcommand{\R}[2]{\up{15pt}#1\dn{8pt} & \up{15pt}#2\dn{8pt}}
\newcommand{\F}[2]{\up{22pt}#1\dn{14pt} & \up{22pt}#2\dn{14pt}}
% A fraction row whose name takes two lines: the two lines straddle the bar.
\newcommand{\W}[2]{\up{16pt}#1 & \up{22pt}#2\dn{14pt}}
\newcommand{\rs}{\\\hline}
% A label grouping the rows below it, spanning both cells.
\newcommand{\grp}[1]{\multicolumn{2}{@{}l@{}}{\rule{0pt}{16pt}\small\bfseries\color{muted}#1}\\}

\hs{Decisions and Trade}
\begin{sheettable}
\R{Net benefit}{$\text{benefit} - \text{direct cost}$} \rs
\R{Opportunity cost}{net benefit of the next best alternative} \rs
\R{Economic cost}{$\text{direct cost} + \text{opportunity cost}$} \rs
\R{Economic rent}{$\text{net benefit} - \text{opportunity cost}$} \rs
\W{Opportunity cost of one unit}{$\dfrac{\text{other good given up}}{\text{this good gained}}$} \\
\end{sheettable}

\hs{Costs}
\begin{sheettable}
\F{Average cost}{$\text{AC} = \dfrac{C(Q)}{Q}$} \rs
\F{Marginal cost}{$\text{MC} = \dfrac{\Delta C}{\Delta Q} = C(Q+1) - C(Q)$} \\
\end{sheettable}

\hs{Elasticity}
\begin{sheettable}
\F{Percentage change}{$\%\,\Delta = \dfrac{\text{new} - \text{old}}{\text{old}} \times 100$} \rs
\F{Price elasticity}{$\varepsilon = -\dfrac{\%\,\Delta\text{ demand}}{\%\,\Delta\text{ price}} = -\dfrac{P}{Q} \times \dfrac{\Delta Q}{\Delta P}$} \\
\F{}{$\phantom{\varepsilon} = -\dfrac{P}{Q} \times \dfrac{1}{\text{slope}}$} \rs
\F{Income elasticity}{$\varepsilon_{\text{income}} = \dfrac{\%\,\Delta\text{ demand}}{\%\,\Delta\text{ income}}$} \rs
\F{Cross-price elasticity}{$\varepsilon_{\text{cross}} = \dfrac{\%\,\Delta\text{ demand for A}}{\%\,\Delta\text{ price of B}}$} \rs
\F{Markup rule}{$\dfrac{P - \text{MC}}{P} = \dfrac{1}{\varepsilon}$} \\
\end{sheettable}
\end{minipage}\hfill
\begin{minipage}[t]{0.475\linewidth}


\hs{Revenue and Profit}
\begin{sheettable}
\R{Revenue}{$R = P \times Q$} \rs
\R{Profit}{$R - C(Q)$} \rs
\F{Marginal revenue}{$\text{MR} = \dfrac{\Delta R}{\Delta Q} = R(Q+1) - R(Q)$} \rs
\R{Profit maximum}{$\text{MR} = \text{MC}$} \\
\end{sheettable}

\hs{Surplus}
\begin{sheettable}
\grp{On one sale}
\R{Buyer's gain}{$\text{WTP} - P$} \rs
\R{Firm's gain}{$P - \text{MC}$} \rs
\R{Joint surplus}{$\text{WTP} - \text{MC}$} \\
\grp{In the market}
\R{Consumer surplus}{$\text{CS} = \tfrac{1}{2} \times Q^* \times (P_0 - P^*)$} \rs
\R{Producer surplus}{$\text{PS} = (P^* - \text{MC}) \times Q^*$} \rs
\R{Deadweight loss}{$\text{DWL} = \tfrac{1}{2} \times (Q_F - Q^*) \times (P^* - \text{MC})$} \\
\grp{Areas}
\R{Triangle}{$\tfrac{1}{2} \times \text{base} \times \text{height}$} \rs
\R{Rectangle}{$\text{base} \times \text{height}$} \\
\end{sheettable}
